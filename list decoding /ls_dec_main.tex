	\section*{Notations}
	
	\addcontentsline{toc}{section}{\large Notations}
	
	\begin{itemize}
		\item We denote a finite set of symbols, called an \emph{alphabet}, by $\Sigma$. A code $\calc$ of block length $n$ over the alphabet $\Sigma$ is just a subset of $\Sigma^n$.
		
		\item For $n\in\mathbb{N} := \{1,2,\ldots\}$, we define $[n]$ to be the set $\{1,2,\ldots,n\}$.
		
		\item  We denote the $q$-entropy function by $H_q(\rho) := \rho \log_q(q-1) - plog_qp - (1-p)log_q(1-p)$.
		
		\item  Let $q$ be a prime power. Then $\mathbb{F}_q$ denotes the finite field of order $q$.
		
		\item We denote the Hamming distance by $\Delta(\bfx,\bfy) := \#\{i\in[n] | x_i \neq y_i\}$ for $\bfx, \bfy\in\Sigma^n$, and the Hamming Ball by $B(\bfy,e):=\{x\in\Sigma^n | \Delta(\bfx,\bfy)\leq e\}$.
		
		\item For a bivariate polynomial $Q(X,Y)$, the maximum degree of $X$ in any monomial of $Q(X,Y)$ is denoted by $\deg_X(Q)$ and similarly for $Y$, we have $\deg_Y(Q)$.
		
		\vfill
		
		
	\end{itemize}
